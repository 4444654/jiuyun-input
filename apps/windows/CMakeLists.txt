cmake_minimum_required(VERSION 3.21)
project(JiuyunInput LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
file(GLOB TSF_SOURCES CONFIGURE_DEPENDS tsf/*.cpp)
add_library(JiuyunIME SHARED ${TSF_SOURCES} tsf/SampleIME.rc tsf/SampleIME.def)
target_compile_definitions(JiuyunIME PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_options(JiuyunIME PRIVATE /utf-8 /W3)
target_include_directories(JiuyunIME PRIVATE tsf)
target_link_libraries(JiuyunIME PRIVATE ole32 oleaut32 uuid user32 gdi32 advapi32 shell32 shlwapi)
add_executable(JiuyunSettings WIN32 Settings.cpp)
target_compile_definitions(JiuyunSettings PRIVATE UNICODE _UNICODE)
target_compile_options(JiuyunSettings PRIVATE /utf-8)
target_link_libraries(JiuyunSettings PRIVATE user32 advapi32 comdlg32 shell32)
